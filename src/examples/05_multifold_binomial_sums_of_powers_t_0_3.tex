      \tsum_{s=0}^{r}
      \tbinom{k-s+r-0}{k-s+r+1} \KnuthRFoldSum{s}{n}{0}
    \right] \nabla^{k} 0^{m},
    \\
    \KnuthRFoldSum{r+1}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+k+r-1}{k+r+1}

    \right] \nabla^{k} 1^{m},
    \\
    \KnuthRFoldSum{r+1}{n}{m}
