\begin{example}
  For non-negative integers $r,n,m$ we have
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+k+r-1}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{k+r-1-s}{k+r-s}
      \KnuthRFoldSum{s}{n}{0}
    \right]
    \nabla^k 0^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+k+r-2}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{k+r-2-s}{k+r-s}
      \KnuthRFoldSum{s}{n}{0}
    \right]
    \nabla^k 1^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+k+r-3}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{k+r-3-s}{k+r-s}
      \KnuthRFoldSum{s}{n}{0}
    \right]
    \nabla^k 2^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+k+r-4}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{k+r-4-s}{k+r-s}
      \KnuthRFoldSum{s}{n}{0}
    \right]
    \nabla^k 3^m.
  \end{align*}
\end{example}
