\begin{example}
  For integer $0 \le t \le 3$, we have
  \begin{align*}
    \KnuthRFoldSum{1}{n}{3}
    &= 0\!\left[\tbinom{n}{1}-\tbinom{0}{1}\right]
    + 1\!\left[\tbinom{n+1}{2}-\tbinom{1}{2}\right]
    - 6\!\left[\tbinom{n+2}{3}-\tbinom{2}{3}\right]
    + 6\!\left[\tbinom{n+3}{4}-\tbinom{3}{4}\right], \\
    %
    \KnuthRFoldSum{1}{n}{3}
    &= 1\!\left[\tbinom{n-1}{1}-\tbinom{-1}{1}\right]
    + 1\!\left[\tbinom{n}{2}-\tbinom{0}{2}\right]
    + 0\!\left[\tbinom{n+1}{3}-\tbinom{1}{3}\right]
    + 6\!\left[\tbinom{n+2}{4}-\tbinom{2}{4}\right], \\
    %
    \KnuthRFoldSum{1}{n}{3}
    &= 8\!\left[\tbinom{n-2}{1}-\tbinom{-2}{1}\right]
    + 7\!\left[\tbinom{n-1}{2}-\tbinom{-1}{2}\right]
    + 6\!\left[\tbinom{n}{3}-\tbinom{0}{3}\right]
    + 6\!\left[\tbinom{n+1}{4}-\tbinom{1}{4}\right], \\
    %
    \KnuthRFoldSum{1}{n}{3}
    &= 27\!\left[\tbinom{n-3}{1}-\tbinom{-3}{1}\right]
    + 19\!\left[\tbinom{n-2}{2}-\tbinom{-2}{2}\right]
    + 12\!\left[\tbinom{n-1}{3}-\tbinom{-1}{3}\right]
    + 6\!\left[\tbinom{n}{4}-\tbinom{0}{4}\right].
  \end{align*}
\end{example}
