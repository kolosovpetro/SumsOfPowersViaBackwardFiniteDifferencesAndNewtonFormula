    t=1:\quad \KnuthRFoldSum{1}{n}{3}
    &=
    n \cdot 1
    + \tfrac{1}{2}(-n+n^2)\cdot 1
    + \tfrac{1}{6}(-n+n^3)\cdot 0

    t=2:\quad \KnuthRFoldSum{1}{n}{3}
    &=
    n \cdot 8
    + \tfrac{1}{2}(-3n+n^2)\cdot 7
    + \tfrac{1}{6}(2n-3n^2+n^3)\cdot 6
    + \tfrac{1}{24}(2n-n^2-2n^3+n^4)\cdot 6,

    t=3:\quad \KnuthRFoldSum{1}{n}{3}
    &=

    + \tfrac{1}{2}(-5n+n^2)\cdot 19
    + \tfrac{1}{6}(11n-6n^2+n^3)\cdot 12 \\

\end{example}
