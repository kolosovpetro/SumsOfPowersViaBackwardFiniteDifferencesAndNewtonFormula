\begin{example}
  For non-negative integers $r,n,m$ we have
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+r-0}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{r-s}{k+r-s}
      \tbinom{s+n-1}{s}
    \right]
    \Delta^k 0^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+r-1}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{r-s-1}{k+r-s}
      \tbinom{s+n-1}{s}
    \right]
    \Delta^k 1^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+r-2}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{r-s-2}{k+r-s}
      \tbinom{s+n-1}{s}
    \right]
    \Delta^k 2^m,
    \\
    \KnuthRFoldSum{r}{n}{m}
    &=
    \tsum_{k=0}^{m}
    \left[
      \tbinom{n+r-3}{k+r}
      -
      \tsum_{s=0}^{r-1}
      \tbinom{r-s-3}{k+r-s}
      \tbinom{s+n-1}{s}
    \right]
    \Delta^k 3^m.
  \end{align*}
\end{example}
