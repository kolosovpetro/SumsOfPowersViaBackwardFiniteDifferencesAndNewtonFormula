The following lemma relates Stirling numbers of the second kind $\stirlingii{n}{k}$
to backward differences of powers $\nabla^{k} t^{m}$.
\begin{lemma} [Stirling Backward Differences]
  For non-negative integers $m$ and $k$, and an arbitrary integer $t$, the following identity holds
  \label{lem:stirling-backward-differencess}
  \begin{align*}
    \nabla^{k} t^{m} = \sum_{j=k}^{m} \binom{t-k}{j-k} \stirlingii{m}{j} j!.
  \end{align*}
  \begin{proof}
    The identity $t^m  = \sum_{k=0}^{m} \binom{t}{k} \stirlingii{m}{k} k!$,
    together with the binomial recurrence
    $\binom{n+1}{k} = \binom{n}{k} + \binom{n}{k-1}$, gives the result.
  \end{proof}
\end{lemma}
The lemma gives the following representation.
\begin{proposition} [Multifold Stirling Sums of Powers]
  \label{prop:multifold-stirling-sums-of-powers}
  For non-negative integers $r$, $n$, and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{r+1}{n}{m}
    =
    \sum_{k=0}^{m}
    \sum_{j=k}^{m}
    \left[
      \binom{n+k+r-t}{k+r+1}
      -
      \sum_{s=0}^{r}
      \binom{k-s+r-t}{k-s+r+1} \binom{s+n-1}{s}
    \right] \binom{t-k}{j-k} \stirlingii{m}{j} j!
  \end{align*}
\end{proposition}
