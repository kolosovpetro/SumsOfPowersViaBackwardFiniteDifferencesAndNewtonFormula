We have Lemma which connects Stirling numbers of the second kind $\stirlingii{n}{k}$
with backward differences of powers $\nabla^{k} t^{m}$
\begin{lemma} [Stirling Backward differences]
  For non-negative integers $m,k$, and an arbitrary integer $t$
  \label{lem:stirling-backward-differencess}
  \begin{align*}
    \nabla^{k} t^{m} = \sum_{j=k}^{m} \binom{t-k}{j-k} \stirlingii{m}{j} j!
  \end{align*}
  \begin{proof}
    By the identity $t^m  = \sum_{k=0}^{m} \binom{t}{k} \stirlingii{m}{k} k!$,
    and by using
    binomial recurrence $\binom{n+1}{k} = \binom{n}{k} + \binom{n}{k-1}$.
  \end{proof}
\end{lemma}
