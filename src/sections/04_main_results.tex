The transition to a formula for sums of powers is straightforward:
\begin{align*}
  \KnuthRFoldSum{1}{n}{m}
  = \sum_{k=0}^{m} \left[ \sum_{j=1}^{n} \binom{j-t+k-1}{k} \right] \nabla^{k} t^m.
\end{align*}
Setting $r=0$ in Lemma~\eqref{lem:backward-hockey-stick-identity}
therefore gives the following closed form for ordinary sums of powers.
\begin{proposition}[Ordinary Sums of Powers]
  \label{prop:ordinary-sums-of-powers}
  For non-negative integers $n$ and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m}
    = \sum_{k=0}^{m}
    \left[
      \binom{n-t+k}{k+1} - \binom{k-t}{k+1}
    \right] \nabla^{k} t^m.
  \end{align*}
\end{proposition}
The basis $\binom{j-t+k-1+r}{k+r}$ is well suited to repeated
applications of Lemma~\eqref{lem:backward-hockey-stick-identity}.
Indeed, setting the parameter $r$ gives a closed form for sums of powers.

In the preceding example, some binomial coefficients have negative upper
indices, which may look unfamiliar.
These coefficients are well defined because they satisfy
the upper negation identity for binomial coefficients
$\binom{-n}{k} = (-1)^{k} \binom{k+n-1}{k}$
in Lemma~\eqref{lem:upper-binomial-negation}.
Thus,
\begin{align*}
  \binom{k-t}{k+1} = \binom{-(t+k)}{k+1}
  = (-1)^{k+1} \binom{(k+1)+(t-k)-1}{k+1}
  = (-1)^{k+1} \binom{t}{k+1}
\end{align*}
\begin{corollary}[Ordinary Sums of Powers]
  \label{cor:ordinary-sums-of-powers-upper-negation}
  For non-negative integers $n$ and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m}
    &= \sum_{k=0}^{m}
    \left[
      \binom{n-t+k}{k+1} + (-1)^{k+2} \binom{t}{k+1}
    \right] \nabla^{k} t^m
    \\
    &= \sum_{k=0}^{m}
    \left[
      \binom{n-t+k}{k+1} + (-1)^{k} \binom{t}{k+1}
    \right] \nabla^{k} t^m.
  \end{align*}
\end{corollary}

We derive a formula for double sums of powers similarly.
Setting $r=1$ in Lemma~\eqref{lem:backward-hockey-stick-identity}
yields the following result.
\begin{proposition}[Double Sums of Powers]
  \label{prop:double-sums-of-powers}
  For non-negative integers $n$ and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{2}{n}{m}
    = \sum_{k=0}^{m}
    \left[
      \binom{n-t+k+1}{k+2}
      - \binom{k-t+1}{k+2} \KnuthRFoldSum{0}{n}{0}
      - \binom{k-t+0}{k+1} \KnuthRFoldSum{1}{n}{0}
    \right] \nabla^{k} t^m.
  \end{align*}
\end{proposition}
The formula includes the correction terms $\KnuthRFoldSum{0}{n}{0}=1$
and $\KnuthRFoldSum{1}{n}{0}=n$.

Similarly, repeated application of Lemma~\eqref{lem:backward-hockey-stick-identity}
yields a closed formula for multifold sums of powers.
\begin{theorem} [Multifold Sums of Powers]
  \label{thm:multifold-sums-of-powers}
  For non-negative integers $r$, $n$, and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{r+1}{n}{m}
    =
    \sum_{k=0}^{m}
    \left[
      \binom{n+k+r-t}{k+r+1}
      -
      \sum_{s=0}^{r}
      \binom{k-s+r-t}{k-s+r+1} \KnuthRFoldSum{s}{n}{0}
    \right] \nabla^{k} t^{m}.
  \end{align*}
\end{theorem}

Using Lemma~\eqref{lem:multifold-sum-of-zero-powers},
we obtain the following binomial form of Theorem~\eqref{thm:multifold-sums-of-powers}.
\begin{proposition} [Multifold Binomial Sums of Powers]
  \label{prop:multifold-binomial-sums-of-powers}
  For non-negative integers $r$, $n$, and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{r+1}{n}{m}
    =
    \sum_{k=0}^{m}
    \left[
      \binom{n+k+r-t}{k+r+1}
      -
      \sum_{s=0}^{r}
      \binom{k-s+r-t}{k-s+r+1} \binom{s+n-1}{s}
    \right] \nabla^{k} t^{m}.
  \end{align*}
\end{proposition}
