The following lemma relates Eulerian numbers $\eulerian{n}{k}$ to backward
differences of powers $\nabla^{k} t^{m}$.
\begin{lemma} [Eulerian Backward Differences]
  For non-negative integers $m$ and $k$, and an arbitrary integer $t$, the following identity holds
  \label{lem:eulerian-backward-differences}
  \begin{align*}
    \nabla^{k} t^{m} = \sum_{j=0}^{m} \binom{t+j-k}{m-k} \eulerian{m}{j}.
  \end{align*}
  \begin{proof}
    Worpitzky's identity
    $t^{m} = \sum_{k=0}^{m} \binom{t+k}{m} \eulerian{m}{k}$,
    together with the binomial recurrence
    $\binom{n+1}{k} = \binom{n}{k} + \binom{n}{k-1}$,
    gives the result. See~\cite{Worpitzky1883}.
  \end{proof}
\end{lemma}
The lemma gives the following representation.
\begin{proposition} [Multifold Eulerian Sums of Powers]
  \label{prop:multifold-eulerian-sums-of-powers}
  For non-negative integers $r$, $n$, and $m$, and an arbitrary integer $t$, the following identity holds
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    =
    \sum_{k=0}^{m}
    \sum_{j=0}^{m}
    \left[
      \binom{n-t+k+r-1}{k+r}
      -
      \sum_{s=0}^{r-1}
      \tbinom{k-t+r-1-s}{k+r-s} \tbinom{s+n-1}{s}
    \right] \binom{t+j-k}{m-k} \eulerian{m}{j}.
  \end{align*}
\end{proposition}
