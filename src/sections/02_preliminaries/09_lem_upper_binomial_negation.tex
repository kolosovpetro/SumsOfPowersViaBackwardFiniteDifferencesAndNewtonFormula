\begin{lemma}[Upper Binomial Negation]
  \label{lem:upper-binomial-negation}
  For integers $n$ and $k$, the following identity holds
  \begin{align*}
    \binom{n}{k} &= (-1)^{k} \binom{k-n-1}{k}, \\
    \binom{-n}{k} &= (-1)^{k} \binom{k+n-1}{k}.
  \end{align*}
  \begin{proof}
    See~\cite[p. 164, Equation (5.14)]{graham1994concrete}.
  \end{proof}
\end{lemma}
