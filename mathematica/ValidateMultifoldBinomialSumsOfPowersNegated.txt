(* ================================================================ *)
(*  Validates Proposition: Multifold Binomial Sums of Powers         *)
(*  (Negated, Backward Difference Version)                           *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  BackwardDifference,
  MultifoldBinomialSumsNegated,
  ValidateMultifoldBinomialSumsNegated
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers (recursive definition)                 *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[r_, n_, m_] := 0 /; n <= 0;

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[
    KnuthRFoldSum[r - 1, k, m],
    {k, 1, n}
  ] /; r > 0;

(* --------------------------------------------------------------- *)
(*  Backward finite difference                                      *)
(*      ∇^k t^m = Σ (-1)^j Binomial[k,j] (t-j)^m                    *)
(* --------------------------------------------------------------- *)

BackwardDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t - j)^m,
    {j, 0, k}
  ];

(* --------------------------------------------------------------- *)
(*  Proposition                                                     *)
(* --------------------------------------------------------------- *)

MultifoldBinomialSumsNegated[r_, n_, m_, t_] :=
  Sum[
    (
      Binomial[n + t + k + r - 1, k + r]
      -
      Sum[
        Binomial[k + t + r - 1 - s, k + r - s]
          Binomial[s + n - 1, s],
        {s, 0, r - 1}
      ]
    )
    BackwardDifference[t, m, k],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                              *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsNegated[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r, n, m]
        -
        MultifoldBinomialSumsNegated[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

(* --------------------------------------------------------------- *)
(*  Run test                                                        *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsNegated[5]
