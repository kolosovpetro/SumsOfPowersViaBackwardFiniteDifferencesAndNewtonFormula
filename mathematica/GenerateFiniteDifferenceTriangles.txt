(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Backward finite difference                                      *)
(*                                                                  *)
(*  ∇^k t^m = Σ (-1)^j Binomial[k,j] (t-j)^m                        *)
(* --------------------------------------------------------------- *)

BackwardDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t - j)^m,
    {j, 0, k}
  ];


(* ================================================================ *)
(*  Compute sequences                                               *)
(* ================================================================ *)
  
Do[

  Print["------------------------------------------------------------"];
  Print["t = ", t];
  Print["------------------------------------------------------------"];

  Print["Flatten format:"];
  Table[BackwardDifference[t, m, k], {m,0,10}, {k,0,m}] //Flatten // Print;

  Print[""],

  {t, 0, 4}

];
