(* ================================================================ *)
(*  Definitions                                                     *)
(* ================================================================ *)

ClearAll[n,L, ForwardDifference];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Binomial coefficient appearing in the decomposition             *)
(*  of prop:multifold-binomial-sums-of-powers                       *)
(* --------------------------------------------------------------- *)

L[n_, t_, r_, k_] := Binomial[n - t + k + r -1, k + r] - Sum[Binomial[k-t+r-1-s, k + r - s] * Binomial[s + n - 1, s], {s, 0, r - 1}];

(* --------------------------------------------------------------- *)
(*  Backward finite difference                                      *)
(*                                                                  *)
(*  ∇^k t^m = Σ (-1)^j Binomial[k,j] (t-j)^m                        *)
(* --------------------------------------------------------------- *)

BackwardDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t - j)^m,
    {j, 0, k}
  ];


(* ================================================================ *)
(*  Parameters                                                      *)
(* ================================================================ *)

m = 3;
r = 1;


(* ================================================================ *)
(*  Compute coefficients and forward differences                    *)
(* ================================================================ *)

Do[

  Print["------------------------------------------------------------"];
  Print["m = ", m, ",   t = ", t];
  Print["------------------------------------------------------------"];

  Print["L[n,t,r,k]:"];
  Table[
    Together @ Expand @ L[n, t, r, k],
    {k, 0, m}
  ] // Print;

  Print["Backward differences:"];
  Table[
    BackwardDifference[t, m, k],
    {k, 0, m}
  ] // Print;

  Print[""],

  {t, 0, 4}

];
