(* ================================================================ *)
(*  Validates Proposition: Multifold Binomial Sums of Powers       *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  BackwardDifference,
  RHS,
  ValidateMultifoldBinomialSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1                                             *)
(* --------------------------------------------------------------- *)

Unprotect[Power];
Power[x_, 0] := 1;
Protect[Power];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                        *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] /; r > 0 :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}];

(* --------------------------------------------------------------- *)
(*  Backward difference operator (as defined in identity)   *)
(* --------------------------------------------------------------- *)

BackwardDifference[t_, m_, k_] :=
  Sum[(-1)^j Binomial[k, j] (t - j)^m, {j, 0, k}];

(* --------------------------------------------------------------- *)
(*  Main RHS expression                                            *)
(* --------------------------------------------------------------- *)

RHS[r_, n_, m_, t_] :=
  Sum[
    (Binomial[n + k + r - t, k + r + 1] - Sum[Binomial[k - s + r - t, k - s + r + 1] Binomial[s + n - 1, s], {s, 0, r}])
    BackwardDifference[t, m, k],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                              *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsOfPowers[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r + 1, n, m] - RHS[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

ValidateMultifoldBinomialSumsOfPowers[5]
