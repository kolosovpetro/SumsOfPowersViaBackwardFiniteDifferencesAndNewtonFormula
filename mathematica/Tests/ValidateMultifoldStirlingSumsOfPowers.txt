(* ================================================================ *)
(*  Validates Proposition: Multifold Stirling sums of powers        *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  ValidateMultifoldStirlingSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                  *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                       *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}];

(* --------------------------------------------------------------- *)
(*  Main identity kernel                                           *)
(* --------------------------------------------------------------- *)

MultifoldStirlingSumsOfPowers[r_, n_, m_, t_] :=
  Sum[
    Sum[
      (
        Binomial[n + k + r - t, k + r + 1]
        - Sum[
            Binomial[k - s + r - t, k - s + r + 1]*
              Binomial[s + n - 1, s],
            {s, 0, r}
          ]
      )
      * Binomial[t - k, j - k]
      * StirlingS2[m, j]
      * j!,
      {j, k, m}
    ],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                             *)
(* --------------------------------------------------------------- *)

ValidateMultifoldStirlingSumsOfPowers[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r + 1, n, m]
        -
        MultifoldStirlingSumsOfPowers[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

(* --------------------------------------------------------------- *)
(*  Run test                                                       *)
(* --------------------------------------------------------------- *)

ValidateMultifoldStirlingSumsOfPowers[5]
