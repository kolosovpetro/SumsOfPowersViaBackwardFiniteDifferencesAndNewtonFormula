(* ================================================================ *)
(*  Validates Corollary: Ordinary Sums of Powers                    *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  BackwardDifference,
  OrdinarySumsOfPowersUpperNegation,
  ValidateOrdinarySumsOfPowersCorollary
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

KnuthRFoldSum[0, n_, m_] := n^m;
KnuthRFoldSum[r_, n_, m_] :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}] /; r > 0;

BackwardDifference[t_, m_, k_] :=
  Sum[(-1)^j Binomial[k, j] (t - j)^m, {j, 0, k}];

OrdinarySumsOfPowersUpperNegation[n_, m_, t_] :=
  Sum[
    (Binomial[n - t + k, k + 1] + (-1)^k Binomial[t, k + 1])
      BackwardDifference[t, m, k],
    {k, 0, m}
  ];

ValidateOrdinarySumsOfPowersCorollary[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[1, n, m] - OrdinarySumsOfPowersUpperNegation[n, m, t]
      ],
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

ValidateOrdinarySumsOfPowersCorollary[10]
