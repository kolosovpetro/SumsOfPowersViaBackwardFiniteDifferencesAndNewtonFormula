  For non-negative integers $r,n,m$, and an arbitrary integer $t$
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}

      \binom{r-s-t}{k+r-s}
      \binom{s+n-1}{s}
    \right] \Delta^{k} t^{m}.
  \end{align*}
\end{proposition}
